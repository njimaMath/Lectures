\handout{2}{有理関数の積分}

\keybox{到達目標：分母の因子から部分分数の形を決め、
対数・べき乗・逆正接の積分に帰着できる。以下、$C$ は積分定数を表す。}

\topic{有理関数と計算の方針}
実係数多項式 $P,Q$（$Q\not\equiv0$）の比 $P(x)/Q(x)$ を有理関数という。
積分は $Q(x)\ne0$ となる区間ごとに考え、区間が異なれば積分定数も別々に選べる。
$\deg P\ge\deg Q$ なら、除法で商 $H$ と余り $R$ を求め、
$$
 P=HQ+R,\qquad \frac{P}{Q}=H+\frac{R}{Q}
 \quad(R=0\ \text{または}\ \deg R<\deg Q)
$$
と書く。商 $H$ を積分し、真分数 $R/Q$ の分母を因数分解して部分分数に分ける。
最後に各項を積分し、微分して確かめよう。

\topic{一次因子の基本積分}
$\alpha\in\mathbb R$、$n\ge1$ を整数とする。$x\ne\alpha$ の区間上で
$$
 \int\frac{dx}{(x-\alpha)^n}=
 \begin{cases}
  \log|x-\alpha|+C,&n=1,\\[3pt]
  -\dfrac{1}{(n-1)(x-\alpha)^{n-1}}+C,&n\ge2.
 \end{cases}
$$
対数になるのは $n=1$ の場合だけである。絶対値は $x<\alpha$ の区間でも必要となる。

\topic{二次因子：判別式と平方完成}
$q(x)=x^2+ax+b$（$a,b\in\mathbb R$）、$D=a^2-4b$ とおく。
$D>0$ なら異なる実根をもつので一次因子に分け、
$D=0$ なら $q(x)=(x+a/2)^2$ として上の公式を使う。
$D<0$ なら既約二次式であり、$c=\sqrt{b-a^2/4}>0$ とおくと
$$
 q(x)=(x+a/2)^2+c^2,\qquad
 \int\frac{dx}{q(x)}=\frac1c\arctan\frac{x+a/2}{c}+C.
$$
分子が一次式 $ux+v$（$u,v\in\mathbb R$）なら、
$ux+v=\tfrac u2q'(x)+(v-ua/2)$ と分けて
$$
 \int\frac{ux+v}{q(x)}\,dx
 =\frac u2\log q(x)+\frac{v-ua/2}{c}\arctan\frac{x+a/2}{c}+C
 \qquad(D<0).
$$
ここでは $q(x)>0$ なので対数の絶対値を省ける。
一般には $\int q'(x)/q(x)\,dx=\log|q(x)|+C$ である。

\topic{部分分数分解：分母から形を決める}
$Q$ を一次因子と既約二次因子の積に分解し、真分数 $R/Q$ に対して、
異なる因子ごとに次の項を用意する（$m\ge1$ は重複度。$m=1$ では最初の項のみ）。
$$
 \begin{array}{c|l}
  Q(x)\text{ の因子}&\text{対応する部分分数}\\
  \hline
  (x-\alpha)^m&\displaystyle
  \frac{A_1}{x-\alpha}+\cdots+\frac{A_m}{(x-\alpha)^m}\\[9pt]
  q(x)^m\ (D<0)&\displaystyle
  \frac{B_1x+E_1}{q(x)}+\cdots+\frac{B_mx+E_m}{q(x)^m}
 \end{array}
$$
係数 $A_k,B_k,E_k$（$1\le k\le m$）は実数であり、この分解は一意に存在する。
重複因子では $1$ 乗から最大のべきまで省かず、二次因子の分子は一次式にする。
分母を払った多項式の恒等式に根を代入し、残りは係数を比較して求める。

\examplelabel{例：重複する一次因子。}
$$
 \frac1{x(x-1)^2}=\frac A x+\frac B{x-1}+\frac E{(x-1)^2}
 \quad\Longleftrightarrow\quad
 1=A(x-1)^2+Bx(x-1)+Ex.
$$
$x=0,1$ より $A=E=1$、$x^2$ の係数より $B=-1$。したがって各区間 $x\ne0,1$ で
$$
 \int\frac{dx}{x(x-1)^2}=\log|x|-\log|x-1|-\frac1{x-1}+C.
$$

\topic{計算例：除法と既約二次因子}
\examplelabel{除法から始める例。}
商の項も忘れずに積分する。
$$
 \frac{x^2+1}{x(x-1)}=1-\frac1x+\frac2{x-1},\qquad
 \int\frac{x^2+1}{x(x-1)}\,dx=x-\log|x|+2\log|x-1|+C.
$$
\examplelabel{一次因子と二次因子の例。}
分母を払って係数を求めると
$$
 \frac{x^2+x+1}{(x-1)(x^2+1)}
 =\frac{3}{2(x-1)}+\frac{-x+1}{2(x^2+1)},
$$
$$
 \int\frac{x^2+x+1}{(x-1)(x^2+1)}\,dx
 =\frac32\log|x-1|-\frac14\log(x^2+1)+\frac12\arctan x+C.
$$

\topic{重複する既約二次因子：帰納公式}
$c>0$、整数 $m\ge1$ に対し、積分定数をそろえるため
$I_m(t)=\int_0^t(s^2+c^2)^{-m}\,ds$（$t\in\mathbb R$）と定める。
$$
 I_1(t)=\frac1c\arctan\frac t c,\qquad
 I_m(t)=\frac{t}{2(m-1)c^2(t^2+c^2)^{m-1}}
       +\frac{2m-3}{2(m-1)c^2}I_{m-1}(t)\quad(m\ge2).
$$
この式は $t/(t^2+c^2)^{m-1}$ を微分して得られる。特に $m=2$ では
$$
 \int\frac{dt}{(t^2+c^2)^2}
 =\frac{t}{2c^2(t^2+c^2)}+\frac1{2c^3}\arctan\frac t c+C.
$$
一般の $q(x)=(x+a/2)^2+c^2$ でも、$t=x+a/2$ と置けば
$$
 \int\frac{ux+v}{q(x)^m}\,dx
 =-\frac{u}{2(m-1)q(x)^{m-1}}
  +\left(v-\frac{ua}{2}\right)I_m(x+a/2)+C\quad(m\ge2).
$$
\keybox{まとめ：有理関数の原始関数は、有理関数・対数・逆正接の有限個の和になる。
答えは微分して確かめ、分母の零点を除いた区間を確認しよう。}

\handouttitle{第2回\quad 演習問題の解答}
{\small 以下、$C$ は積分定数であり、分母の零点で区切られる区間ごとに異なってよい。}\par
\begin{problems}
\setlength{\itemsep}{7pt}
\item $x<2$ または $x>2$ の各区間で
$$
 \int\frac{dx}{x-2}=\log|x-2|+C.
$$
$x<2$ では $\log|x-2|=\log(2-x)$ だから、微分すると
$-1/(2-x)=1/(x-2)$ となり、この区間でも正しい。

\item $x<-1$ または $x>-1$ の各区間で、$n=3$ の公式より
$$
 \int\frac{dx}{(x+1)^3}=-\frac1{2(x+1)^2}+C.
$$
べき乗の公式 $-1/\{(n-1)(x-\alpha)^{n-1}\}$ に $n=1$ を代入すると
分母が0になる。$n=1$ の場合は別に計算し、$\log|x-\alpha|+C$ を用いる。

\item どちらの分母もすべての実数 $x$ で正なので、区間は $\mathbb R$ でよい。
\text{(a)} $x^2+2x+5=(x+1)^2+4$ より
$$
 \int\frac{dx}{x^2+2x+5}=\frac12\arctan\frac{x+1}{2}+C.
$$
\text{(b)} $q(x)=x^2+x+3=(x+\tfrac12)^2+\tfrac{11}{4}$ とおく。
$x+2=\tfrac12q'(x)+\tfrac32$ と分けると
$$
 \int\frac{x+2}{x^2+x+3}\,dx
 =\frac12\log(x^2+x+3)+\frac3{\sqrt{11}}\arctan\frac{2x+1}{\sqrt{11}}+C.
$$

\item \text{(a)} 分母を払った恒等式は
$$
 1=A(x+1)^2+Bx(x+1)+Ex.
$$
$x=0,-1$ の代入と係数比較から $A=1$、$B=E=-1$ を得る。したがって
$$
 \frac1{x(x+1)^2}=\frac1x-\frac1{x+1}-\frac1{(x+1)^2},\qquad
 \int\frac{dx}{x(x+1)^2}=\log|x|-\log|x+1|+\frac1{x+1}+C.
$$
区間は $(-\infty,-1)$、$(-1,0)$、$(0,\infty)$ のいずれかである。
\text{(b)} 恒等式と係数は
$$
 x+1=A(x^2+1)+(Bx+E)x,\qquad A=1,\quad B=-1,\quad E=1.
$$
よって $x<0$ または $x>0$ の各区間で
$$
 \frac{x+1}{x(x^2+1)}=\frac1x+\frac{-x+1}{x^2+1},\qquad
 \int\frac{x+1}{x(x^2+1)}\,dx
 =\log|x|-\frac12\log(x^2+1)+\arctan x+C.
$$

\item 除法により $x^3=x(x^2+1)-x$ だから、$\mathbb R$ 上で
$$
 \int\frac{x^3}{x^2+1}\,dx
 =\int\left(x-\frac{x}{x^2+1}\right)dx
 =\frac{x^2}{2}-\frac12\log(x^2+1)+C.
$$

\item 帰納公式に $c=2$、$m=2$ を代入すると
$$
 I_2(t)=\frac{t}{8(t^2+4)}+\frac18 I_1(t)
 =\frac{t}{8(t^2+4)}+\frac1{16}\arctan\frac t2.
$$
したがって、$I_2(0)=0$ と $\arctan1=\pi/4$ より
$$
 \int_0^2\frac{dx}{(x^2+4)^2}=I_2(2)=\frac1{32}+\frac\pi{64}
 =\frac{2+\pi}{64}>0.
$$
\end{problems}
