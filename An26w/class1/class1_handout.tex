\handout{1}{逆関数と逆三角関数}

\topic{逆関数の定義}
関数$f$に対し、次を満たすとき$g$を$f$の逆関数と呼ぶ:
$$
 {\ f(g(y))=y\quad(y\in B),\qquad g(f(x))=x\quad(x\in A)\ }
$$
ここで$A$は$f$の定義域、$B$は$g$の定義域。式を求めるには、$y=f(x)$ を $x$ について解けばよい。

\topic{逆関数の微分公式}
$y=f(x)$、$x=g(y)$ とする。$f$ が微分可能で $f'(g(y))\ne0$ ならば、
$f(g(y))=y$ の両辺を $y$ で微分して
$$
 f'(g(y))g'(y)=1,\qquad
 {\ g'(y)=\frac{1}{f'(g(y))}\ }.
$$
逆関数のグラフでは横と縦が入れ替わるため、その傾きは元の傾きの逆数になる。

\topic{逆三角関数の定義}
三角関数の定義域を一対一になる区間に制限して、その逆関数を定める。
$$
\begin{array}{c|c|c|c}
\text{逆関数}&\text{定義域}&\text{値域}&\text{定義}\\
\hline
\arcsin y&[-1,1]&[-\pi/2,\pi/2]&\sin(\arcsin y)=y\\
\arccos y&[-1,1]&[0,\pi]&\cos(\arccos y)=y\\
\arctan y&\mathbb R&(-\pi/2,\pi/2)&\tan(\arctan y)=y
\end{array}
$$
たとえば $\arcsin(1/2)=\pi/6$ であり、$5\pi/6$ は値域の外なので主値ではない。

\topic{逆三角関数の微分と積分}
$$
\begin{aligned}
(\arcsin x)'&=\frac1{\sqrt{1-x^2}} &&(-1<x<1),&
\int\frac{dx}{\sqrt{1-x^2}}&=\arcsin x+C,\\
(\arccos x)'&=-\frac1{\sqrt{1-x^2}} &&(-1<x<1),&
\int\frac{-dx}{\sqrt{1-x^2}}&=\arccos x+C,\\
(\arctan x)'&=\frac1{1+x^2} &&(x\in\mathbb R),&
\int\frac{dx}{1+x^2}&=\arctan x+C.
\end{aligned}
$$
\keybox{確認点：逆関数では式だけでなく定義域と値域を確認する。平方根の符号は、逆三角関数の値域から決まる。}

\topic{演習問題}
\begin{problems}
\item $f(x)=3x-5$ の逆関数 $g$ を求めよ.
\item $f(x)=x^2+2x$、$x\ge-1$ の逆関数とその定義域を求めよ。
\item $\arcsin(-\sqrt3/2)$、$\arccos(-1/2)$、$\arctan(1)$ を求めよ。
\item $y=\arccos x$ の微分を導け。
\item $\displaystyle\int_0^{1/2}\frac{dx}{\sqrt{1-x^2}}$ を求めよ。
\end{problems}
