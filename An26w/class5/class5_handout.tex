\handout{5}{広義積分}
\keybox{到達目標：1変数・2変数の広義積分を極限で定め、収束・発散を判断できる。}

\topic{特異点や無限遠を極限で扱う}
有限端点では $[a,b)$、無限遠では $[a,\infty)$ の各閉部分区間で可積分な実数値関数 $f$ を考える。対応する広義積分を
$$
 \int_a^bf(x)\,dx:=\lim_{t\to b-}\int_a^tf(x)\,dx,
 \qquad
 \int_a^\infty f(x)\,dx:=\lim_{R\to\infty}\int_a^Rf(x)\,dx.
$$
と定める。有限の極限が存在するとき「収束する」という。内部の特異点 $c\in(a,b)$ は左右に分け、
両方の広義積分がそれぞれ収束する場合だけ和を取る。

\topic{べき関数の広義積分}
$p\in\mathbb R$ に対して、特異点付近と無限遠では条件が逆になる。
$$
 \int_0^1x^{-p}\,dx=
 \begin{cases}\dfrac1{1-p},&p<1,\\+\infty,&p\ge1,\end{cases}
 \qquad
 \int_1^\infty x^{-p}\,dx=
 \begin{cases}\dfrac1{p-1},&p>1,\\+\infty,&p\le1.\end{cases}
$$


\topic{定理：比較による収束・発散の判定}
$a\in\mathbb R$、$a<b\le+\infty$ とし、実数値関数 $f,g$ を考える。\\

\examplelabel{発散の判定：下から押さえる。}
$$
 f(x)\ge |g(x)|,\quad \int_a^b|g(x)|\,dx=+\infty
 \quad\Longrightarrow\quad \int_a^bf(x)\,dx=+\infty.
$$
\examplelabel{収束の判定：絶対値を上から押さえる。}
$$
 |f(x)|\le g(x),\quad \int_a^bg(x)\,dx<\infty
 \quad\Longrightarrow\quad \int_a^b|f(x)|\,dx<\infty.
$$
このとき $\int_a^bf(x)\,dx$ は有限の値として存在する。$a=-\infty$の場合も同様。


\topic{非負関数の広義重積分}
$D\subset\mathbb R^2$ 上の連続な非負関数 $f$ を考える。$D_n\subset D$ を有界な積分可能領域で、
$D_n\subset D_{n+1}$、$\bigcup_nD_n=D$ を満たすものとする。このとき、
$$
 \iint_Df\,dx\,dy:=\lim_{n\to\infty}\iint_{D_n}f\,dx\,dy\in[0,+\infty].
$$
積分値は単調に増え、極限は領域列の取り方によらない。有限なら収束、$+\infty$ なら発散という。
例えば $D=(0,1]^2$ では $D_n=[1/n,1]^2$、$D=\mathbb R^2$ では $D_n=[-n,n]^2$ と取れる。

\topic{比較判定の例}
\examplelabel{符号が変わっても収束。}
$x\ge1$ では
$$
 \left|\frac{\sin x}{1+x^2}\right|\le x^{-2},\qquad
 \int_1^\infty x^{-2}\,dx=1<\infty.
$$
したがって $\int_1^\infty\frac{\sin x}{1+x^2}\,dx$ は収束する。

\examplelabel{特異点での発散。}
$0<x\le1$ では
$$
 \frac{1+x}{x}\ge x^{-1},\qquad \int_0^1x^{-1}\,dx=+\infty.
$$
したがって $\int_0^1\frac{1+x}{x}\,dx=+\infty$。

\topic{計算例：特異点を避ける・無限遠へ広げる}
\noindent\begin{minipage}[t]{0.73\linewidth}
\vspace{0pt}
\examplelabel{両軸に近づく例。}
$D=(0,1]^2$ で $f(x,y)=1/\sqrt{xy}$ は非負であり、$x=0$ と $y=0$ に近づくと大きくなる。
$0<\varepsilon<1$ に対し $D_\varepsilon=[\varepsilon,1]^2$ とすると
$$
 \begin{aligned}
 \iint_{D_\varepsilon}\frac{dx\,dy}{\sqrt{xy}}
 &=\left(\int_\varepsilon^1t^{-1/2}\,dt\right)^2
 =4(1-\sqrt\varepsilon)^2,\\
 \iint_D\frac{dx\,dy}{\sqrt{xy}}&=\lim_{\varepsilon\downarrow0}4(1-\sqrt\varepsilon)^2=4.
 \end{aligned}
$$

\examplelabel{無限領域の例。}
$Q=[0,\infty)^2$ では $Q_R=[0,R]^2$ を広げる。非負性と $e^{-x-y}=e^{-x}e^{-y}$ より
$$
 \iint_Qe^{-x-y}\,dx\,dy
 =\lim_{R\to\infty}\left(\int_0^Re^{-t}\,dt\right)^2
 =\lim_{R\to\infty}(1-e^{-R})^2=1.
$$

\exercises
\begin{problems}
\item \examplelabel{極限から計算する。}
$\displaystyle\int_0^1\frac{dx}{\sqrt{x}}$、$\displaystyle\int_1^\infty\frac{dx}{x^{3/2}}$ を、
それぞれ下端を $\varepsilon>0$、上端を $R>1$ にして計算し、極限と値を求めよ。
\problemspace

\item \examplelabel{べき関数と比較する。}
(a) $\displaystyle\int_0^1\frac{dx}{\sqrt{x}(1+x)}$、
(b) $\displaystyle\int_1^\infty\frac{dx}{\sqrt{x^2+x}}$ の収束・発散を判定せよ。
(b) では $x^2+x\le2x^2$（$x\ge1$）を使う。
\problemspace

\item \examplelabel{絶対値で比較する。}
$\displaystyle\int_1^\infty\frac{\cos x}{x^{3/2}}\,dx$ が絶対収束することを示せ。
また、積分値の絶対値が $2$ 以下であることを示せ。
\problemspace


\item \examplelabel{長方形で近づける。}
$D=(0,1]^2$ とする。$D_\varepsilon=[\varepsilon,1]^2$（$0<\varepsilon<1$）で積分してから、
$\displaystyle\iint_D\frac{y}{\sqrt{x}}\,dx\,dy$ の収束と値を調べよ。
\problemspace

\item \examplelabel{無限領域へ広げる。}
$Q=[1,\infty)^2$ を $Q_R=[1,R]^2$（$R>1$）で近づけ、
(a) $\displaystyle\iint_Q\frac{dx\,dy}{x^2y^2}$、
(b) $\displaystyle\iint_Q\frac{dx\,dy}{xy}$ の収束・発散と、収束する場合の値を調べよ。
\problemspace
\item \examplelabel{特異点で分ける。}
$\displaystyle\int_{-1}^1\frac{dx}{x}$ の収束・発散を左右別々に調べよ。
$\displaystyle\int_{-1}^{-\varepsilon}\frac{dx}{x}+\int_\varepsilon^1\frac{dx}{x}=0$（$0<\varepsilon<1$）でも、収束といえないのはなぜか。（AIに聞くこと推奨）
\problemspace

\end{problems}
