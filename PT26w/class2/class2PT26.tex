\handouttitle{第2回\quad 確率変数と確率分布}

\topic{確率変数と可測性}
$(\Omega,\F,P)$ を確率空間、$\mathcal B(\R)$ を実数の開集合から生成される
最小の $\sigma$-加法族（ボレル集合族）とする。写像 $X:\Omega\to\R$ が
確率変数であるとは、次の可測性を満たすことである。
$$
  X^{-1}(B):=\{\omega\in\Omega:X(\omega)\in B\}\in\F
  \qquad\bigl(B\in\mathcal B(\R)\bigr).
$$
可測性は $\{X\le x\}\in\F$ がすべての $x\in\R$ について成り立つことと同値である。

\topic{確率分布：観測値の空間上の確率測度}
$X$ の分布（押し出し測度）$\mu_X$ を、$\mathcal B(\R)$ 上で
$$
  \mu_X(B)=P(X\in B):=P\bigl(X^{-1}(B)\bigr)
  \qquad\bigl(B\in\mathcal B(\R)\bigr)
$$
と定める。離散確率空間では、各値 $x\in\R$ の確率は
$$
  P(X=x)=\sum_{\omega\in\Omega:\,X(\omega)=x}p_\omega.
$$


\topic{分布関数}
$F_X(x):=P(X\le x)$ と定める。次の性質をもつ。
$$
  \begin{aligned}
    x\le y&\ \Longrightarrow\ F_X(x)\le F_X(y),
    &\lim_{h\downarrow0}F_X(x+h)&=F_X(x),\\
    \lim_{x\to-\infty}F_X(x)&=0,
    &\lim_{x\to+\infty}F_X(x)&=1.
  \end{aligned}
$$
特に $a<b$ に対して $P(a<X\le b)=F_X(b)-F_X(a)$ である。

\examplelabel{例：公平なコインを独立に2回投げ、表の個数を表す確率変数は次で定義される:}
$$
  X(H,H)=2,\quad X(H,T)=X(T,H)=1,\quad X(T,T)=0,
$$
$$
  P(X=0)=\frac14,\qquad P(X=1)=\frac12,\qquad P(X=2)=\frac14.
$$
したがって $F_X(0)=1/4$、$F_X(1)=3/4$、$F_X(2)=1$ となる。

\topic{演習問題}
\begin{enumerate}[label=\arabic*.]
  \item 公平なサイコロを独立に2回投げ、出た目を $U,V$ とする。
  $X=U+V$、$Y=|U-V|$ の分布を求め、すべての $x\in\R$ に対する
  分布関数 $F_X(x)$、$F_Y(x)$ を求めよ。
  \item $\Omega=[0,1]$、$\F=\mathcal B([0,1])$、$P=\lambda$（ルベーグ測度）とする。
  $X(\omega)=3\omega+1$、$Y(\omega)=(X(\omega)-5/2)^2$ の分布関数を求めよ。
  $Y$ の計算では、事象 $\{Y\le y\}$ を $\omega$ の区間として表せ。
\end{enumerate}

\topic{レポート問題}
\begin{enumerate}[label=\arabic*.]
  \item 公平なコインを３回投げる。裏の数を表す確率変数を定義したい。確率空間およびその確率変数の定義を述べよ.
  \item 確率空間$(\Omega,\mathcal F,\mathbb P)$と写像 $X:\Omega\to \mathbb R$を考える. 次の集合族が$\sigma$加法族であることを示せ:
$$\{B\subset \mathbb R:~X^{-1}(B) \in \mathcal F\}.$$
\end{enumerate}

\handouttitle{第3回\quad 期待値とルベーグ積分}

\topic{有限和から非負確率変数の積分へ}
以下、$(\Omega,\F,P)$ を確率空間とする。$A\in\F$ の指示関数 $\mathbf{1}_A$ は
$A$ 上で1、それ以外で0をとる。非負単関数 $Z=\sum_{j=1}^m a_j\mathbf{1}_{A_j}$
（$a_j\ge0$、$A_j\in\F$）について
$$
  \E[Z]:=\sum_{j=1}^m a_jP(A_j)
$$
と定める。非負確率変数 $X$、つまり$X:\Omega\to [0,\infty)$、について
$$
  \E[X]:=\int_\Omega X\,dP
  =\sup\{\E[Z]:0\le Z\le X,\ Z\text{ は非負単関数}\}\in[0,\infty]
$$
と定義する。

\topic{可積分性と期待値の基本性質}
$X^+:=\max\{X,0\}$、$X^-:=\max\{-X,0\}$ とおく。
$$
  X=X^+-X^-,\qquad |X|=X^++X^-,\qquad
  \E[|X|]=\E[X^+]+\E[X^-].
$$
$\E[|X|]<\infty$ のとき $X$ は可積分であるといい、$\E[X]:=\E[X^+]-\E[X^-]$ と定める。
可積分な $X,Y$ と $a,b\in\R$ に対して
$$
  \E[aX+bY]=a\E[X]+b\E[Y],\qquad
  X\le Y\ \Longrightarrow\ \E[X]\le\E[Y].
$$

\topic{極限と期待値の交換}
\begingroup
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
\examplelabel{単調収束定理：}
$$
  0\le X_n\uparrow X\quad\text{a.s.}
  \quad\Longrightarrow\quad
  \E[X_n]\uparrow\E[X].
$$

\examplelabel{優収束定理：}
$$
    X_n\to X,\quad |X_n|\le |Y|,\quad\E[|Y|]<\infty
  \quad\Longrightarrow\quad
  \begin{gathered}
    \E[X_n]\to\E[X].
  \end{gathered}
$$
\par\endgroup

\topic{演習問題}
\begin{enumerate}[label=\arabic*.]
  \item 重み付きサイコロ $P(\{k\})=k/21$（$k=1,\ldots,6$）で、出た目を $X$ とする。
  $\E[X]$、$\E[X^2]$ を求め、$a\in\R$ に対して $\E[(X-a)^2]$ を最小にする
  $a$ と、その最小値を求めよ。
  \item $\Omega=[0,1]$、$\F=\mathcal B([0,1])$、$P=\lambda$（ルベーグ測度）とする。
  $X(\omega)=\omega(1-\omega)$ の期待値 $\E[X]$ を求めよ。
\end{enumerate}

\topic{レポート問題}
以下、$\Omega=(0,1)$、$\F=\mathcal B((0,1))$、$P=\lambda$ とする。
\begin{enumerate}[label=\arabic*.]
  \item $X(\omega)=\omega^2$ とする。$\E[X]$ を積分により求めよ。
  \item $X_n(\omega)=n\mathbf{1}_{(0,1/n)}(\omega)$ とする。
  $X_n\to0$ と $\E[X_n]=1$ を示し、優収束定理を適用できない理由を述べよ。
\end{enumerate}
