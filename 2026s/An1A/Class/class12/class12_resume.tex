\section*{小テスト}

\subsection*{問題}

\begin{enumerate}
\item
$$
f(x,y) = \begin{cases}
    \dfrac{x^4 y^4}{x^2+y^2} & (x,y)\neq (0,0) \\
    0 & (x,y)=(0,0)
\end{cases}
$$
が $(0,0)$で全微分可能かどうか判定せよ。
\item
$f(x,y)=x|x|+y|y|$ について $\partial_x f$ を求めよ.
\end{enumerate}

\subsection*{解答}

\begin{enumerate}
\item
まず偏導関数を定義から求める。
$$
f_x(0,0)=\lim_{h\to0}\frac{f(h,0)-f(0,0)}{h}=\lim_{h\to0}\frac{0}{h}=0,
$$
同様に $f_y(0,0)=0$。

全微分可能性を確かめるため，
$$
\lim_{(h,k)\to(0,0)}\frac{f(h,k)-f(0,0)-0\cdot h-0\cdot k}{\sqrt{h^2+k^2}}
=\lim_{(h,k)\to(0,0)}\frac{h^4k^4}{(h^2+k^2)^{3/2}}
$$
を調べる。$r=\sqrt{h^2+k^2}$ とおくと $|h|\le r$，$|k|\le r$ なので
$$
0\le\frac{h^4k^4}{(h^2+k^2)^{3/2}}\le\frac{r^8}{r^3}=r^5\to0\quad(r\to0).
$$
よって $f$ は $(0,0)$ で全微分可能であり，$df_{(0,0)}=0$ である。

\item
一変数関数 $g(t)=t|t|$ を考える。$t>0$ では $g'(t)=2t$，$t<0$ では
$g'(t)=-2t$ である。また
$$
g'(0)=\lim_{h\to0}\frac{h|h|-0}{h}=\lim_{h\to0}|h|=0.
$$
よって $g'(t)=2|t|$ がすべての $t$ で成り立つ。$f(x,y)=g(x)+g(y)$ なので
$$
\partial_x f(x,y)=g'(x)=2|x|.
$$
\end{enumerate}

\section*{1. 極値と停留点}

\begin{definition}
$f:\R^2\to\R$ とする。点 $(a,b)$ の近くのすべての点 $(x,y)$ について
$$
f(x,y)\ge f(a,b)
$$
が成り立つとき，$(a,b)$ を極小点という。同様に，近くのすべての点について
$$
f(x,y)\le f(a,b)
$$
が成り立つとき，$(a,b)$ を極大点という。極小点と極大点をあわせて極値点という。
\end{definition}

\begin{theorem}[極値を取るための必要条件]
$f$ が $(a,b)$ の近くで定義され，$(a,b)$ で全微分可能であるとする。$(a,b)$ が内点の極値点ならば
$$
f_x(a,b)=0,\qquad f_y(a,b)=0
$$
である。
\end{theorem}

\begin{proof}
$y=b$ と固定して，一変数関数 $g(x)=f(x,b)$ を考える。$(a,b)$ が極値点なら，$g$ は $x=a$ で極値を取る。したがって一変数関数の場合の必要条件より $g'(a)=0$ である。これは $f_x(a,b)=0$ を意味する。

同様に，$x=a$ と固定して $h(y)=f(a,y)$ を考えると $f_y(a,b)=0$ を得る。
\end{proof}

\begin{definition}
$$
f_x(a,b)=0,\qquad f_y(a,b)=0
$$
を満たす点 $(a,b)$ を停留点という。
\end{definition}

\begin{remark}
停留点は極値点の候補である。しかし，停留点が必ず極値点になるわけではない。
\end{remark}

\begin{example}
$$
f(x,y)=x^2+y^2
$$
は $(0,0)$ で極小値を取る。一方，
$$
g(x,y)=x^2-y^2
$$
は $(0,0)$ で停留点をもつが，極大点でも極小点でもない。実際，$x$ 軸方向では $g(x,0)=x^2\ge0$ であり，$y$ 軸方向では $g(0,y)=-y^2\le0$ である。
\end{example}

\section*{2. 二次形式}

\begin{definition}
$$
Q(h,k)=Ah^2+2Bhk+Ck^2
$$
の形の式を2変数の二次形式という。
\end{definition}

二次形式は行列を使って
$$
Q(h,k)
=
\begin{pmatrix}h&k\end{pmatrix}
\begin{pmatrix}A&B\\B&C\end{pmatrix}
\begin{pmatrix}h\\k\end{pmatrix}
$$
と書ける。

\begin{definition}
$(h,k)\ne(0,0)$ に対して常に $Q(h,k)>0$ となるとき，$Q$ は正定値であるという。
常に $Q(h,k)<0$ となるとき，$Q$ は負定値であるという。正の値も負の値も取るとき，不定であるという。
\end{definition}

\begin{theorem}[2変数二次形式の符号判定]
$Q(h,k)=Ah^2+2Bhk+Ck^2$ とし，
$$
D=AC-B^2
$$
とおく。
\begin{itemize}
\item $D>0$ かつ $A>0$ なら，$Q$ は正定値である。
\item $D>0$ かつ $A<0$ なら，$Q$ は負定値である。
\item $D<0$ なら，$Q$ は不定 (極値ではない) である。
\item $D=0$ のときは，この判定だけでは決まらない。
\end{itemize}
\end{theorem}

\begin{proof}
$A\ne0$ のとき，平方完成により
$$
Q(h,k)
=A\left(h+\frac{B}{A}k\right)^2
+\frac{AC-B^2}{A}k^2
$$
と書ける。この式から $D>0$ の場合は $A$ と同じ符号にそろうことがわかる。

$D<0$ の場合は，$k=0$ とすると $Q(h,0)=Ah^2$ であり，また $h=-Bk/A$ と選ぶと
$$
Q\left(-\frac{B}{A}k,k\right)=\frac{D}{A}k^2
$$
となる。この2つは反対の符号を取るので，$Q$ は不定である。
\end{proof}

\begin{example}
$$
Q(h,k)=2h^2+2hk+2k^2
$$
では $A=2, B=1, C=2$ なので
$$
D=AC-B^2=4-1=3>0,\qquad A>0.
$$
したがって $Q$ は正定値である。
\end{example}

\section*{3. 2変数のテイラーの定理}

\begin{theorem}[2次のテイラーの定理]
$f$ が点 $(a,b)$ の近くで $C^2$ 級であるとする。このとき $(h,k)\to(0,0)$ で
$$
\begin{aligned}
f(a+h,b+k)
=&\,f(a,b)+f_x(a,b)h+f_y(a,b)k\\
&+\frac12\left\{
f_{xx}(a,b)h^2+2f_{xy}(a,b)hk+f_{yy}(a,b)k^2
\right\}
+o(h^2+k^2)
\end{aligned}
$$
が成り立つ。
\end{theorem}

\begin{remark}
一次の項は勾配による近似である。停留点では一次の項が消えるので，極値判定では2次の項が主役になる。
\end{remark}

\begin{example}
$$
f(x,y)=\log(1+x+2y)
$$
を $(0,0)$ の近くで2次まで展開する。まず
$$
f(0,0)=0,\qquad f_x(0,0)=1,\qquad f_y(0,0)=2
$$
である。また
$$
f_{xx}(0,0)=-1,\qquad f_{xy}(0,0)=-2,\qquad f_{yy}(0,0)=-4.
$$
したがって
$$
\log(1+h+2k)
=h+2k-\frac12h^2-2hk-2k^2+o(h^2+k^2)
$$
である。
\end{example}

\pointbox{
2次のテイラー展開では
$$
f_{xx}h^2+2f_{xy}hk+f_{yy}k^2
$$
という二次形式が現れる。この二次形式の符号が，停留点の形を決める。
}

\section*{4. 2変数関数の極値判定}

$f$ が $C^2$ 級で，$(a,b)$ が停留点であるとする。ヘッセ行列を
$$
H_f(a,b)=
\begin{pmatrix}
f_{xx}(a,b)&f_{xy}(a,b)\\
f_{xy}(a,b)&f_{yy}(a,b)
\end{pmatrix}
$$
と書く。

\begin{theorem}[2変数関数の極値判定]
$(a,b)$ が停留点であるとし，
$$
D=f_{xx}(a,b)f_{yy}(a,b)-f_{xy}(a,b)^2
$$
とおく。
\begin{itemize}
\item $D>0$ かつ $f_{xx}(a,b)>0$ なら，$(a,b)$ は極小点である。
\item $D>0$ かつ $f_{xx}(a,b)<0$ なら，$(a,b)$ は極大点である。
\item $D<0$ なら，$(a,b)$ は鞍点である。
\item $D=0$ のときは，この判定だけでは決まらない。
\end{itemize}
\end{theorem}

\begin{proof}
$(a,b)$ は停留点なので $f_x(a,b)=f_y(a,b)=0$ である。テイラーの定理より
$$
\begin{aligned}
f(a+h,b+k)-f(a,b)
=&\,\frac12\left\{
f_{xx}(a,b)h^2+2f_{xy}(a,b)hk+f_{yy}(a,b)k^2
\right\}\\
&+o(h^2+k^2)
\end{aligned}
$$
となる。したがって，原点に十分近いところでは，増減は主にこの二次形式の符号で決まる。二次形式が正定値なら極小，負定値なら極大，不定なら方向によって増減が変わるので鞍点である。
\end{proof}

\begin{example}
$$
f(x,y)=x^3+y^3-3xy
$$
の極値を調べる。偏導関数は
$$
f_x=3x^2-3y,\qquad f_y=3y^2-3x
$$
である。停留点は
$$
y=x^2,\qquad x=y^2
$$
を満たす点なので，
$$
(x,y)=(0,0),\ (1,1)
$$
である。

2階偏導関数は
$$
f_{xx}=6x,\qquad f_{xy}=-3,\qquad f_{yy}=6y.
$$
したがって
$$
D=36xy-9
$$
である。$(0,0)$ では $D=-9<0$ なので鞍点である。$(1,1)$ では $D=27>0$ かつ $f_{xx}(1,1)=6>0$ なので極小点である。
\end{example}

\begin{remark}
$D=0$ の場合は注意が必要である。例えば $f(x,y)=x^4+y^4$ は $(0,0)$ で極小値を取るが，ヘッセ行列は零行列である。一方，$g(x,y)=x^3+y^2$ も $(0,0)$ で $D=0$ だが，極小点ではない。
\end{remark}

\section*{5. 一般の $n$ 変数関数の場合}

$f:\R^n\to\R$ とし，点 $\boldsymbol{a}\in\R^n$ の近くで $C^2$ 級であるとする。勾配とヘッセ行列を
$$
\nabla f(\boldsymbol{a})
=\left(\frac{\partial f}{\partial x_1}(\boldsymbol{a}),\ldots,
\frac{\partial f}{\partial x_n}(\boldsymbol{a})\right),
$$
$$
H_f(\boldsymbol{a})
=\left(
\frac{\partial^2 f}{\partial x_i\partial x_j}(\boldsymbol{a})
\right)_{1\le i,j\le n}
$$
と書く。

2次のテイラー展開は
$$
f(\boldsymbol{a}+\boldsymbol{h})
=f(\boldsymbol{a})
+\nabla f(\boldsymbol{a})\cdot\boldsymbol{h}
+\frac12\boldsymbol{h}^{T}H_f(\boldsymbol{a})\boldsymbol{h}
+o(\|\boldsymbol{h}\|^2)
$$
となる。

\begin{theorem}[一般の場合の極値判定]
$\nabla f(\boldsymbol{a})=\boldsymbol{0}$ とする。
\begin{itemize}
\item $H_f(\boldsymbol{a})$ が正定値なら，$\boldsymbol{a}$ は極小点である。
\item $H_f(\boldsymbol{a})$ が負定値なら，$\boldsymbol{a}$ は極大点である。
\item $H_f(\boldsymbol{a})$ が不定なら，$\boldsymbol{a}$ は鞍点である。
\end{itemize}
\end{theorem}

\begin{remark}
一般の $n$ 変数では，正定値性は固有値がすべて正であることと同値である。負定値性は固有値がすべて負であることと同値である。この授業では，計算問題としては2変数の場合の判定をまず確実に使えるようにする。
\end{remark}

\section*{まとめ}

\pointbox{
極値判定の流れ:
\begin{itemize}
\item 偏導関数を計算し，停留点を求める。
\item 2階偏導関数を計算し，ヘッセ行列を作る。
\item 2変数なら $D=f_{xx}f_{yy}-f_{xy}^2$ を見て，極小・極大・鞍点を判定する。
\end{itemize}
}
